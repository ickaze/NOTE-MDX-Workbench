package main

import (
	"fmt"
	"strings"
)

type voiceDefinition struct {
	Bytes      []uint8 `json:"bytes"`
	ID         int     `json:"id"`
	File       string  `json:"file"`
	Line       int     `json:"line"`
	EndLine    int     `json:"endLine"`
	ToneOffset int     `json:"toneOffset"`
}

// The compiler supplies the last valid definition's line; this avoids matching @ commands or comments.
func appendVoiceDefinition(text, voice string, anchorEnd int) (string, error) {
	text = normalizeSourceText(text)
	voice = strings.TrimSpace(normalizeSourceText(voice))
	lines := strings.Split(text, "\n")
	if anchorEnd < 0 || anchorEnd > len(lines) {
		return "", fmt.Errorf("invalid instrument insertion line: %d", anchorEnd)
	}
	insertion := "\n" + voice + "\n"
	if anchorEnd == 0 {
		anchorEnd = len(lines)
	}
	offset := 0
	for i := 0; i < anchorEnd && i < len(lines); i++ {
		offset += len(lines[i])
		if i < len(lines)-1 {
			offset++
		}
	}
	if offset == len(text) && !strings.HasSuffix(text, "\n") {
		insertion = "\n" + insertion
	}
	result := text[:offset] + insertion + text[offset:]
	return ensureOverwriteDirective(result), nil
}

func ensureOverwriteDirective(text string) string {
	lines := strings.Split(text, "\n")
	// Directives inside comments do not enable overwrite.
	overwrite, title := false, -1
	firstSignificant := -1
	listed := true
	block := false
	for i, line := range lines {
		var visible strings.Builder
		for j := 0; j < len(line); j++ {
			if block {
				if j+1 < len(line) && line[j:j+2] == "*/" {
					block = false
					j++
				}
				continue
			}
			if j+1 < len(line) && line[j:j+2] == "/*" {
				block = true
				j++
				continue
			}
			if line[j] == ';' || (j+1 < len(line) && line[j:j+2] == "//") {
				break
			}
			visible.WriteByte(line[j])
		}
		fields := strings.Fields(strings.ToLower(strings.TrimSpace(visible.String())))
		if len(fields) > 0 {
			if firstSignificant < 0 {
				firstSignificant = i
			}
			if len(line) == 0 || line[0] == ' ' || line[0] == '\t' {
				continue
			}
			if fields[0] == "#nlist" {
				listed = false
				continue
			}
			if fields[0] == "#list" {
				listed = true
				continue
			}
			if !listed {
				continue
			}
			if fields[0] == "#overwrite" {
				overwrite = true
			}
			if fields[0] == "#title" && i == firstSignificant {
				title = i
			}
		}
	}
	result := text
	if !overwrite {
		if title < 0 {
			result = "#overwrite\n" + result
		} else {
			// Insertion after a #title must use the original line offset, before voice additions below it.
			at := 0
			for i := 0; i <= title; i++ {
				at += len(lines[i])
				if i < len(lines)-1 {
					at++
				}
			}
			if title == len(lines)-1 && !strings.HasSuffix(text, "\n") {
				result = result[:at] + "\n#overwrite\n" + result[at:]
			} else {
				result = result[:at] + "#overwrite\n" + result[at:]
			}
		}
	}
	return result
}

func voiceWithOffsetScope(voice string, offset int) string {
	if offset == 0 {
		return voice
	}
	return fmt.Sprintf("#toneofs 0\n%s\n#toneofs %d", strings.TrimSpace(normalizeSourceText(voice)), offset)
}
