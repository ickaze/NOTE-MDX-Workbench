func mmlTokenEnd(text []uint16, begin int) int {
	if begin < 0 || begin >= len(text) {
		return begin
	}
	end := begin + 1
	if text[begin] == '$' || text[begin] == '#' {
		for end < len(text) && ((text[end] >= 'A' && text[end] <= 'Z') || (text[end] >= 'a' && text[end] <= 'z') || (text[end] >= '0' && text[end] <= '9') || text[end] == '-') {
			end++
		}
		return end
	}
	for end < len(text) {
		cursor := end
		for cursor < len(text) && (text[cursor] == ' ' || text[cursor] == '\t') {
			cursor++
		}
		if cursor == len(text) {
			break
		}
		char := text[cursor]
		if (char >= '0' && char <= '9') || char == '+' || char == '-' || char == '=' || char == '"' || char == '.' || char == '%' || char == '^' || char == '~' || char == '&' || (text[begin] == 'n' && char == ',') {
			end = cursor + 1
		} else {
			break
		}
	}
	return end
}
