package main
import("bufio";"fmt";"os";"strconv";"time")
var app struct{lastSrcTime,lastPdxTime time.Time}
type fixtureInfo struct{stamp time.Time}
func(f fixtureInfo)ModTime()time.Time{return f.stamp}
func fixtureStat(s string)(fixtureInfo,error){if s=="missing"{return fixtureInfo{},fmt.Errorf("missing")};n,e:=strconv.ParseInt(s,10,64);return fixtureInfo{time.Unix(0,n)},e}
func observe(src,pdx string)bool{
	st, err := fixtureStat(src)
	if err != nil {
		return false
	}
	changed := false
	if !app.lastSrcTime.IsZero() && !st.ModTime().Equal(app.lastSrcTime) {
		changed = true
	}
	app.lastSrcTime = st.ModTime()
	if pdx != "" {
		if ps, e := fixtureStat(pdx); e == nil {
			if !app.lastPdxTime.IsZero() && ps.ModTime().After(app.lastPdxTime) {
				changed = true
			}
			app.lastPdxTime = ps.ModTime()
		}
	}
 return changed
}
