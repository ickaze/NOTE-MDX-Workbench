cmake_minimum_required(VERSION 3.16)
project(IndependentOPM VERSION 0.14.0 LANGUAGES C CXX)
set(INDEPENDENT_OPM_VERSION "0.14")
add_library(independent_opm src/ym2151.cpp)
target_include_directories(independent_opm PUBLIC include)
target_compile_features(independent_opm PUBLIC cxx_std_17)
add_executable(opm_render src/main.cpp)
target_link_libraries(opm_render PRIVATE independent_opm)
add_executable(opm_tests tests/test_core.cpp)
target_link_libraries(opm_tests PRIVATE independent_opm)
enable_testing()
add_test(NAME core COMMAND opm_tests)

add_executable(opm_precision_tests tests/test_precision.cpp)
target_link_libraries(opm_precision_tests PRIVATE independent_opm)
add_test(NAME precision COMMAND opm_precision_tests)

add_executable(opm_output_timing_tests tests/test_output_timing.cpp)
target_link_libraries(opm_output_timing_tests PRIVATE independent_opm)
add_test(NAME output_timing COMMAND opm_output_timing_tests)

add_executable(opm_random_tests tests/test_random.cpp)
target_link_libraries(opm_random_tests PRIVATE independent_opm)
add_test(NAME measured_random COMMAND opm_random_tests)

add_executable(opm_sequence_tests tests/test_sequence.cpp)
target_link_libraries(opm_sequence_tests PRIVATE independent_opm)
add_test(NAME inferred_sequence COMMAND opm_sequence_tests)

add_executable(opm_periodic_tests tests/test_periodic.cpp)
target_link_libraries(opm_periodic_tests PRIVATE independent_opm)
add_test(NAME measured_periodic COMMAND opm_periodic_tests)

option(IOPM_BUILD_DLL "Build the C ABI shared library" ON)
if(IOPM_BUILD_DLL)
    add_library(independent_opm_dll SHARED src/dll_api.cpp src/ym2151.cpp)
    target_compile_features(independent_opm_dll PRIVATE cxx_std_17)
    target_compile_definitions(independent_opm_dll PRIVATE IOPM_BUILD_DLL)
    target_include_directories(independent_opm_dll PUBLIC include)
    # Avoid clashing with the existing static core's independent_opm.lib.
    set_target_properties(independent_opm_dll PROPERTIES OUTPUT_NAME independent_opm
        ARCHIVE_OUTPUT_NAME independent_opm_import CXX_VISIBILITY_PRESET hidden)
    if(WIN32)
        target_sources(independent_opm_dll PRIVATE dll/independent_opm.def)
    endif()
    add_executable(opm_c_api_test tests/test_c_api.c)
    target_link_libraries(opm_c_api_test PRIVATE independent_opm_dll)
    add_test(NAME c_api COMMAND opm_c_api_test)
    add_executable(opm_dll_example examples/dll_client.c)
    target_link_libraries(opm_dll_example PRIVATE independent_opm_dll)
endif()
if(IOPM_BUILD_DLL)
    add_executable(opm_dll_equivalence tests/test_dll_equivalence.cpp)
    target_link_libraries(opm_dll_equivalence PRIVATE independent_opm independent_opm_dll)
    add_test(NAME dll_equivalence COMMAND opm_dll_equivalence)
endif()

add_executable(opm_key_on_tests tests/test_key_on.cpp)
target_link_libraries(opm_key_on_tests PRIVATE independent_opm)
add_test(NAME key_on_mapping COMMAND opm_key_on_tests)

if(IOPM_BUILD_DLL)
    add_executable(opm_state_tests tests/test_state.cpp)
    target_link_libraries(opm_state_tests PRIVATE independent_opm independent_opm_dll)
    add_test(NAME state_serialization COMMAND opm_state_tests)
endif()

add_executable(opm_noise_level_tests tests/test_noise_level.cpp)
target_link_libraries(opm_noise_level_tests PRIVATE independent_opm)
add_test(NAME measured_noise_level COMMAND opm_noise_level_tests)

if(IOPM_BUILD_DLL)
    add_executable(opm_channel_tests tests/test_channels.cpp)
    target_link_libraries(opm_channel_tests PRIVATE independent_opm_dll)
    add_test(NAME channel_render COMMAND opm_channel_tests)
