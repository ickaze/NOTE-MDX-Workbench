cmake_minimum_required(VERSION 3.20)
project(NOTE_MDX_Workbench VERSION 2.10.0 LANGUAGES C CXX)
if(WIN32 AND NOT MSVC)
    message(FATAL_ERROR "Windows builds require the MSVC toolchain (Visual Studio 2022).")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/utf-8 /W4 /EHsc)
    add_compile_definitions(NOMINMAX)
endif()
add_library(workbench_zlib STATIC native/zlib/adler32.c native/zlib/crc32.c
    native/zlib/deflate.c native/zlib/trees.c native/zlib/zutil.c)
target_include_directories(workbench_zlib PUBLIC native/zlib)
add_library(frontend_logic STATIC native/frontend/text_logic.cpp native/frontend/output_pipeline.cpp native/frontend/source_watch.cpp)
target_link_libraries(frontend_logic PUBLIC workbench_zlib)
target_include_directories(frontend_logic PUBLIC native/frontend)
add_library(notemdx_core STATIC native/notemdx/src/compiler.cpp)
target_include_directories(notemdx_core PUBLIC native/notemdx/include)
add_library(mdx_core STATIC native/src/formats.cpp native/src/player.cpp native/src/vgm_export.cpp native/src/mrl_export.cpp)
target_include_directories(mdx_core PUBLIC native/src native/independent-opm/include)
if(WIN32)
    enable_language(RC)
    add_executable(workbench_gui WIN32
        native/gui/main.cpp native/gui/language.cpp native/gui/settings.cpp
        native/gui/document.cpp native/gui/runtime.cpp native/gui/output.cpp
        native/gui/controls.cpp native/gui/editor.cpp native/gui/monitor.cpp native/gui/smoke.cpp assets/app.rc)
    target_compile_definitions(workbench_gui PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(workbench_gui PRIVATE assets)
    target_link_libraries(workbench_gui PRIVATE frontend_logic comctl32 comdlg32 user32 gdi32 winmm ole32)
    set_target_properties(workbench_gui PROPERTIES OUTPUT_NAME "NOTE-MDX-Workbench")
    if(MSVC)
        target_link_options(workbench_gui PRIVATE /STACK:8388608 $<$<CONFIG:Release>:/DEBUG:NONE>)
    endif()
    add_library(workbench_core SHARED native/src/workbench_bridge.cpp
        native/frontend/frontend_bridge.cpp
        native/src/pdx_editor_embed.cpp
        native/pdxeditor/src/App.cpp native/pdxeditor/src/Audio.cpp
        native/pdxeditor/src/Pdx.cpp native/pdxeditor/src/Settings.cpp)
    target_compile_definitions(workbench_core PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(workbench_core PRIVATE frontend_logic notemdx_core mdx_core winmm comdlg32 user32 gdi32 comctl32 msimg32 shell32 shlwapi ole32 uuid)
    add_custom_command(TARGET workbench_core POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:workbench_core>/pdx_languages"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${CMAKE_CURRENT_SOURCE_DIR}/native/pdxeditor/languages/ja.lang"
            "${CMAKE_CURRENT_SOURCE_DIR}/native/pdxeditor/languages/en.lang"
            "$<TARGET_FILE_DIR:workbench_core>/pdx_languages")
    set_target_properties(workbench_core PROPERTIES PREFIX "")
    if(MSVC)
        target_link_options(workbench_core PRIVATE $<$<CONFIG:Release>:/DEBUG:NONE>)
    endif()
else()
    target_sources(mdx_core PRIVATE native/independent-opm/src/ym2151.cpp native/independent-opm/src/dll_api.cpp)
endif()
option(WB_ENABLE_REFERENCE_TESTS "Enable optional frozen-Go differential tests" OFF)
include(CTest)
if(BUILD_TESTING)
    add_executable(vgm-export-tests native/tests/vgm_export_tests.cpp)
    target_link_libraries(vgm-export-tests PRIVATE mdx_core notemdx_core)
    if(WIN32)
        target_link_libraries(vgm-export-tests PRIVATE winmm)
    endif()
    if(MSVC)
        target_link_options(vgm-export-tests PRIVATE /STACK:8388608)
    endif()
    add_test(NAME vgm-export COMMAND vgm-export-tests)
    add_executable(gui-json-tests native/tests/gui_json_tests.cpp)
    add_test(NAME gui-json COMMAND gui-json-tests)
    find_package(Threads REQUIRED)
    add_executable(source-watch-test native/frontend/watch_runner.cpp)
    target_link_libraries(source-watch-test PRIVATE frontend_logic)
    add_executable(latest-worker-tests native/tests/latest_worker_tests.cpp)
    target_include_directories(latest-worker-tests PRIVATE native/frontend)
    target_link_libraries(latest-worker-tests PRIVATE Threads::Threads)
    add_test(NAME latest-worker COMMAND latest-worker-tests)
    set_tests_properties(latest-worker PROPERTIES TIMEOUT 20)
    add_executable(output-pipeline-test native/frontend/output_runner.cpp)
    target_link_libraries(output-pipeline-test PRIVATE frontend_logic)
    add_executable(frontend-compat native/frontend/compat_runner.cpp)
    target_link_libraries(frontend-compat PRIVATE frontend_logic)
    add_library(frontend-bridge-test SHARED native/frontend/frontend_bridge.cpp)
    target_link_libraries(frontend-bridge-test PRIVATE frontend_logic)
    set_target_properties(frontend_logic PROPERTIES POSITION_INDEPENDENT_CODE ON)
    set_target_properties(workbench_zlib PROPERTIES POSITION_INDEPENDENT_CODE ON)
    if(NOT WIN32 AND WB_ENABLE_REFERENCE_TESTS)
        find_package(Python3 COMPONENTS Interpreter QUIET)
        find_program(WB_ORACLE_GO go)
        if(Python3_Interpreter_FOUND AND WB_ORACLE_GO)
            add_test(NAME source-watch
                COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/native/tests/migration/compare_watch.py"
                    --cpp "$<TARGET_FILE:source-watch-test>" --go "${WB_ORACLE_GO}"
                    --report "${CMAKE_CURRENT_BINARY_DIR}/watch-comparison.json")
            add_test(NAME output-pipeline
                COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/native/tests/migration/compare_output.py"
                    --cpp "$<TARGET_FILE:output-pipeline-test>" --go "${WB_ORACLE_GO}"
                    --report "${CMAKE_CURRENT_BINARY_DIR}/output-comparison.json")
        endif()
        find_program(WB_ORACLE_GO go)
        if(Python3_Interpreter_FOUND AND WB_ORACLE_GO)
            add_test(NAME frontend-differential
                COMMAND "${Python3_EXECUTABLE}"
                    "${CMAKE_CURRENT_SOURCE_DIR}/native/tests/migration/compare_helpers.py"
                    --cpp "$<TARGET_FILE:frontend-compat>"
                    --bridge "$<TARGET_FILE:frontend-bridge-test>"
                    --go "${WB_ORACLE_GO}"
                    --report "${CMAKE_CURRENT_BINARY_DIR}/frontend-comparison.json")
        endif()
    endif()
    add_executable(compiler-tests native/tests/notemdx_tests.cpp)
    target_link_libraries(compiler-tests PRIVATE notemdx_core)
    add_test(NAME compiler COMMAND compiler-tests)
    add_executable(selection-tests native/tests/selection_tests.cpp)
    target_link_libraries(selection-tests PRIVATE notemdx_core)
    add_test(NAME selection COMMAND selection-tests)
    add_executable(core-tests native/tests/core_tests.cpp)
    target_link_libraries(core-tests PRIVATE mdx_core)
    if(WIN32)
        target_link_libraries(core-tests PRIVATE winmm)
    endif()
    if(MSVC)
        target_link_options(core-tests PRIVATE /STACK:8388608)
    endif()
    add_test(NAME player COMMAND core-tests)
    add_executable(mapping-tests native/tests/mapping_tests.cpp)
    target_link_libraries(mapping-tests PRIVATE notemdx_core mdx_core)
    if(WIN32)
        target_link_libraries(mapping-tests PRIVATE winmm)
    endif()
    add_test(NAME mapping COMMAND mapping-tests)
    add_executable(mrl-export-tests native/tests/mrl_export_tests.cpp)
    target_link_libraries(mrl-export-tests PRIVATE mdx_core)
    if(WIN32)
        target_link_libraries(mrl-export-tests PRIVATE winmm)
    endif()
    add_test(NAME mrl-export COMMAND mrl-export-tests)
    add_executable(selection-loop-tests native/tests/selection_loop_tests.cpp)
    target_link_libraries(selection-loop-tests PRIVATE notemdx_core mdx_core)
    if(WIN32)
        target_link_libraries(selection-loop-tests PRIVATE winmm)
    endif()
    add_test(NAME selection-loop COMMAND selection-loop-tests)
endif()
